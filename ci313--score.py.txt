"""Re-score payload-free focused and negative/positive acceptance receipts."""
import json
from pathlib import Path
import sys

root = Path(sys.argv[1])
results = {}
nodes = {}
for label in ("focused", "red-base", "green", "red-product-revert"):
    lines = (root / ("resume2-ci313-" + label + ".txt")).read_text().splitlines()
    counts = [json.loads(line.removeprefix("E1B_COUNTS=")) for line in lines if line.startswith("E1B_COUNTS=")]
    codes = [int(line.removeprefix("exit=")) for line in lines if line.startswith("exit=")]
    assert len(counts) == len(codes) == 1
    results[label] = {"counts": counts[0], "exit": codes[0]}
    nodes[label] = {line.split(" ", 1)[1] for line in lines if line.startswith(("PASSED ", "FAILED "))}
assert results["focused"]["counts"]["passed"] == 3 and results["focused"]["exit"] == 0
assert results["green"]["counts"]["passed"] == 62 and results["green"]["exit"] == 0
for label in ("red-base", "red-product-revert"):
    assert results[label]["counts"]["failed"] == 62 and results[label]["exit"] == 1
assert nodes["red-base"] == nodes["green"] == nodes["red-product-revert"] and len(nodes["green"]) == 62
print(json.dumps({"same62_nodeids": True, "results": results}, sort_keys=True))
