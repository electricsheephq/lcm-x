# Optional dependencies for LCM-X semantic/hybrid retrieval (#674).
#
# Local semantic recall is ON by default (#1186): LCM_EMBEDDINGS_ENABLED
# defaults to true with LCM_EMBEDDING_PROVIDER=fastembed and
# LCM_EMBEDDING_MODEL=BAAI/bge-small-en-v1.5. plugin.yaml declares
# `fastembed==0.8.0` under pip_dependencies, so Hermes installs it into the
# environment that runs Hermes and keeps it across `hermes update`.
#
# This file is a FALLBACK only: for a host whose Hermes does not install
# plugin-declared dependencies, or to repair an environment by hand:
#
#     pip install -r requirements-semantic.txt
#
# A package installed by hand is not recorded by Hermes, so a host update can
# drop it. `/lcm doctor` (the embedding_provider_health check) reports an
# unavailable provider and names a missing dependency. No restart is needed
# once the dependency is present: the provider import is lazy and existing
# vectors are untouched. Set LCM_EMBEDDINGS_ENABLED=false to turn semantic
# recall off; LCM-X then needs nothing from this file.
#
# Provider-specific. Only install what LCM_EMBEDDING_PROVIDER selects:
#   fastembed          -> this file (local ONNX, no signup, no daemon)
#   voyage             -> nothing to install; set VOYAGE_API_KEY
#   openai-compatible  -> nothing to install; set the API key + base URL
#   ollama             -> nothing to install; run the Ollama daemon

fastembed>=0.8.0,<1.0
